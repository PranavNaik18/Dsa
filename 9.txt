MERGE AND QUICK SORT



#include <stdio.h>
#include <time.h>

void merge(int a[],int l,int m,int r) {
    int t[100],i=l,j=m+1,k=0;
    while(i<=m&&j<=r)
        t[k++]=(a[i]<a[j])?a[i++]:a[j++];
    while(i<=m) t[k++]=a[i++];
    while(j<=r) t[k++]=a[j++];
    for(i=l,k=0;i<=r;i++) a[i]=t[k++];
}
void ms(int a[],int l,int r) {
    if(l<r) {
        int m=(l+r)/2;
        ms(a,l,m);
        ms(a,m+1,r);
        merge(a,l,m,r);
    }
}
void qs(int a[],int l,int r) {
    int i=l,j=r,p=a[(l+r)/2],t;
    while(i<=j) {
        while(a[i]<p)i++;
        while(a[j]>p)j--;
        if(i<=j) {
            t=a[i];a[i]=a[j];a[j]=t;
            i++;j--;
        }
    }
    if(l<j) qs(a,l,j);
    if(i<r) qs(a,i,r);
}

    int a[100],b[100],n,i;

    printf("Merge Sort time: %f\n",
    (double)(e-s)/CLOCKS_PER_SEC);

    s=clock();
    qs(b,0,n-1);
    e=clock();
    printf("Quick Sort time: %f\n",
    (double)(e-s)/CLOCKS_PER_SEC);

    printf("Sorted: ");
    for(i=0;i<n;i++) printf("%d ",a[i]);
    return 0;
}
