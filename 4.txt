Q) Write a C program to implement Stack using Array with Push, Pop, Peek and Display operations.with queue enqueue and deque

#include <stdio.h>
int main() {
    int s[10],q[10],top=-1,f=-1,r=-1,ch,x,i;
    while(1) {
        printf("\n1.Push 2.Pop 3.StackDisplay");
        printf("\n4.Enqueue 5.Dequeue 6.QueueDisplay 7.Exit\n");
        scanf("%d",&ch);
        switch(ch) {
            case 1:
                scanf("%d",&x);
                if(top==9) printf("Stack Full");
                else s[++top]=x;
                break;
            case 2:
                if(top==-1) printf("Stack Empty");
                else printf("Pop: %d",s[top--]);
                break;
            case 3:
                for(i=top;i>=0;i--) printf("%d ",s[i]);
                break;
            case 4:
                scanf("%d",&x);
                if((r+1)%10==f) printf("Queue Full");
                else {
                    if(f==-1) f=0;
                    r=(r+1)%10;
                    q[r]=x;
                }
                break;
            case 5:
                if(f==-1) printf("Queue Empty");
                else {
                    printf("Dequeue: %d",q[f]);
                    if(f==r) f=r=-1;
                    else f=(f+1)%10;
                }
                break;
            case 6:
                if(f!=-1) {
                    i=f;
                    while(1) {
                        printf("%d ",q[i]);
                        if(i==r) break;
                        i=(i+1)%10;
                    }
                }
                break;
            case 7: return 0;
        }
    }
}
