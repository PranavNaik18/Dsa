linked list using stack and queues 



#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *next;
};

struct node *top=NULL,*f=NULL,*r=NULL;

// Stack Push
void push(int x) {
    struct node *newnode=malloc(sizeof(struct node));
    if(!newnode) return;
    newnode->data=x;
    newnode->next=top;
    top=newnode;
}

// Stack Pop
void pop() {
    if(!top) printf("Stack Empty");
    else {
        struct node *newnode=top;
        printf("Pop: %d",newnode->data);
        top=top->next;
        free(newnode);
    }
}

// Stack Display
void displayStack() {
    struct node *newnode=top;
    if(!top) printf("Stack Empty");
    else {
        printf("Stack: ");
        while(newnode) {
            printf("%d ",newnode->data);
            newnode=newnode->next;
        }
    }
}

// Queue Enqueue
void enqueue(int x) {
    struct node *newnode=malloc(sizeof(struct node));
    if(!newnode) return;
    newnode->data=x;
    newnode->next=NULL;

    if(!r) f=r=newnode;
    else {
        r->next=newnode;
        r=newnode;
    }
}

// Queue Dequeue
void dequeue() {
    if(!f) printf("Queue Empty");
    else {
        struct node *newnode=f;
        printf("Dequeue: %d",newnode->data);
        f=f->next;
        if(!f) r=NULL;
        free(newnode);
    }
}

// Queue Display
void displayQueue() {
    struct node *newnode=f;
    if(!f) printf("Queue Empty");
    else {
        printf("Queue: ");
        while(newnode) {
            printf("%d ",newnode->data);
            newnode=newnode->next;
        }
    }
}

int main() {
    int ch,x;

    while(1) {
        printf("\n\n1.Push 2.Pop 3.Enqueue 4.Dequeue");
        printf("\n5.Stack Display 6.Queue Display 7.Exit\n");
        printf("Enter choice: ");
        scanf("%d",&ch);

        if(ch==1) {
            printf("Enter value: ");
            scanf("%d",&x);
            push(x);
        }
        else if(ch==2) pop();
        else if(ch==3) {
            printf("Enter value: ");
            scanf("%d",&x);
            enqueue(x);
        }
        else if(ch==4) dequeue();
        else if(ch==5) displayStack();
        else if(ch==6) displayQueue();
        else if(ch==7) break;
        else printf("Invalid choice");
    }

    return 0;
}
