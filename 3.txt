5)Develop a C program to map the elements of a 2-D array marks[50][5] into a 1-D array, read marks, display the 1-D array, and retrieve marks using the mapping formula.

#include <stdio.h>

int main()
{
    int marks[50][5];
    int oneD[250];
    int n, i, j, student, subject;
    int index;

    printf("Enter number of students: ");
    scanf("%d", &n);

    printf("Enter marks:\n");

    for(i = 0; i < n; i++)
    {
        for(j = 0; j < 5; j++)
        {
            scanf("%d", &marks[i][j]);

            oneD[i * 5 + j] = marks[i][j];
        }
    }

    printf("\n1-D Array:\n");

    for(i = 0; i < n * 5; i++)
        printf("%d ", oneD[i]);

    printf("\n\nEnter student number (0 to %d): ", n - 1);
    scanf("%d", &student);

    printf("Enter subject number (0 to 4): ");
    scanf("%d", &subject);

    index = student * 5 + subject;

    printf("Marks = %d", oneD[index]);

    return 0;
}







Q)Write a C program to map a 2-D array into a 1-D array using Row-major and Column-major order.

#include <stdio.h>

int main()
{
    int a[10][10], row, col;
    int i, j;

    printf("Enter rows and columns: ");
    scanf("%d %d", &row, &col);

    printf("Enter elements:\n");

    for(i = 0; i < row; i++)
        for(j = 0; j < col; j++)
            scanf("%d", &a[i][j]);

    printf("\nRow Major:\n");

    for(i = 0; i < row; i++)
        for(j = 0; j < col; j++)
            printf("%d ", a[i][j]);

    printf("\n\nColumn Major:\n");

    for(j = 0; j < col; j++)
        for(i = 0; i < row; i++)
            printf("%d ", a[i][j]);

    return 0;
}


1)Store a sparse matrix using 1-D representation. 

#include <stdio.h>

int main()
{
    int a[10][10], sparse[100][3];
    int row, col, i, j, k = 0;

    printf("Enter rows and columns: ");
    scanf("%d %d", &row, &col);

    printf("Enter matrix:\n");

    for(i = 0; i < row; i++)
        for(j = 0; j < col; j++)
            scanf("%d", &a[i][j]);

    for(i = 0; i < row; i++)
    {
        for(j = 0; j < col; j++)
        {
            if(a[i][j] != 0)
            {
                sparse[k][0] = i;
                sparse[k][1] = j;
                sparse[k][2] = a[i][j];
                k++;
            }
        }
    }

    printf("\nRow Column Value\n");

    for(i = 0; i < k; i++)
        printf("%d    %d      %d\n",
               sparse[i][0],
               sparse[i][1],
               sparse[i][2]);

    return 0;
}

2)Implement matrix addition using mapped arrays

#include <stdio.h>

int main()
{
    int a[100], b[100], c[100];
    int row, col, n, i;

    printf("Enter rows and columns: ");
    scanf("%d %d", &row, &col);

    n = row * col;

    printf("Enter first matrix elements:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter second matrix elements:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &b[i]);

    for(i = 0; i < n; i++)
        c[i] = a[i] + b[i];

    printf("Result:\n");

    for(i = 0; i < n; i++)
        printf("%d ", c[i]);

    return 0;
}

4)Extend the program to represent lower triangular matrices efficiently

#include <stdio.h>

int main()
{
    int a[10][10], b[100];
    int n, i, j, k = 0;

    printf("Enter size of matrix: ");
    scanf("%d", &n);

    printf("Enter matrix:\n");

    for(i = 0; i < n; i++)
        for(j = 0; j < n; j++)
            scanf("%d", &a[i][j]);

    for(i = 0; i < n; i++)
        for(j = 0; j <= i; j++)
            b[k++] = a[i][j];

    printf("1-D representation:\n");

    for(i = 0; i < k; i++)
        printf("%d ", b[i]);

    return 0;
}
