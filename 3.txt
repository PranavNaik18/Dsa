5)Develop a C program to map the elements of a 2-D array marks[50][5] into a 1-D array, read marks, display the 1-D array, and retrieve marks using the mapping formula.


#include <stdio.h>

int main()
{
    int marks[50][5];
    int oneD[250];
    int n, i, j, student, subject;
    int index;

    printf("Enter number of students: ");
    scanf("%d", &n);

    printf("Enter marks:\n");

    for(i = 0; i < n; i++)
    {
        for(j = 0; j < 5; j++)
        {
            scanf("%d", &marks[i][j]);
            oneD[i * 5 + j] = marks[i][j];
        }
    }

    printf("\n1-D Array:\n");
    printf("Student\tMarks\n");

    for(i = 0; i < n; i++)
    {
        for(j = 0; j < 5; j++)
            printf("%d\t%d\n", i + 1, oneD[i * 5 + j]);
    }

    printf("\nEnter student number (0 to %d):", n - 1);
    scanf("%d", &student);

    printf("Enter subject number (0 to 4): ");
    scanf("%d", &subject);

    index = student * 5 + subject;

    printf("Marks = %d", oneD[index]);

    return 0;
}
