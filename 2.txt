Q)Write a C program to implement Bubble Sort and Selection Sort and compare their performance.
#include <stdio.h>

int main()
{
    int a[50], b[50];
    int n, i, j, temp, min;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements:\n");

    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
        b[i] = a[i];
    }

    /* Bubble Sort */
    for(i = 0; i < n - 1; i++)
    {
        for(j = 0; j < n - i - 1; j++)
        {
            if(a[j] > a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("\nBubble Sort: ");

    for(i = 0; i < n; i++)
        printf("%d ", a[i]);

    /* Selection Sort */
    for(i = 0; i < n - 1; i++)
    {
        min = i;

        for(j = i + 1; j < n; j++)
        {
            if(b[j] < b[min])
                min = j;
        }

        temp = b[i];
        b[i] = b[min];
        b[min] = temp;
    }

    printf("\nSelection Sort: ");

    for(i = 0; i < n; i++)
        printf("%d ", b[i]);

    return 0;
}
