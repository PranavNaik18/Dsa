Q)Write a C program to implement Bubble Sort and Selection Sort and compare their performance.
#include <stdio.h>

int main()
{
    int a[50], b[50];
    int n, i, j, temp, min;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements:\n");

    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
        b[i] = a[i];
    }

    /* Bubble Sort */
    for(i = 0; i < n - 1; i++)
    {
        for(j = 0; j < n - i - 1; j++)
        {
            if(a[j] > a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("\nBubble Sort: ");

    for(i = 0; i < n; i++)
        printf("%d ", a[i]);

    /* Selection Sort */
    for(i = 0; i < n - 1; i++)
    {
        min = i;

        for(j = i + 1; j < n; j++)
        {
            if(b[j] < b[min])
                min = j;
        }

        temp = b[i];
        b[i] = b[min];
        b[min] = temp;
    }

    printf("\nSelection Sort: ");

    for(i = 0; i < n; i++)
        printf("%d ", b[i]);

    return 0;
}



1)Sort employee records based on salary.

#include <stdio.h>

struct Employee
{
    char name[30];
    float salary;
};

int main()
{
    struct Employee e[50], temp;
    int n, i, j;

    printf("Enter number of employees: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("\nEnter name: ");
        scanf("%s", e[i].name);

        printf("Enter salary: ");
        scanf("%f", &e[i].salary);
    }

    for(i = 0; i < n - 1; i++)
    {
        for(j = i + 1; j < n; j++)
        {
            if(e[i].salary > e[j].salary)
            {
                temp = e[i];
                e[i] = e[j];
                e[j] = temp;
            }
        }
    }

    printf("\nEmployees sorted by salary:\n");

    for(i = 0; i < n; i++)
        printf("%s %.2f\n", e[i].name, e[i].salary);

    return 0;
}


2)Compare Bubble Sort, Selection Sort, and Insertion Sort for random datasets. 

#include <stdio.h>

void bubble(int a[], int n)
{
    int i, j, t;

    for(i = 0; i < n - 1; i++)
        for(j = 0; j < n - i - 1; j++)
            if(a[j] > a[j + 1])
            {
                t = a[j];
                a[j] = a[j + 1];
                a[j + 1] = t;
            }
}

void selection(int a[], int n)
{
    int i, j, min, t;

    for(i = 0; i < n - 1; i++)
    {
        min = i;

        for(j = i + 1; j < n; j++)
            if(a[j] < a[min])
                min = j;

        t = a[i];
        a[i] = a[min];
        a[min] = t;
    }
}

void insertion(int a[], int n)
{
    int i, j, key;

    for(i = 1; i < n; i++)
    {
        key = a[i];
        j = i - 1;

        while(j >= 0 && a[j] > key)
        {
            a[j + 1] = a[j];
            j--;
        }

        a[j + 1] = key;
    }
}

int main()
{
    int a[50], b[50], c[50];
    int n, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements:\n");

    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
        b[i] = a[i];
        c[i] = a[i];
    }

    bubble(a, n);
    selection(b, n);
    insertion(c, n);

    printf("\nBubble: ");
    for(i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\nSelection: ");
    for(i = 0; i < n; i++)
        printf("%d ", b[i]);

    printf("\nInsertion: ");
    for(i = 0; i < n; i++)
        printf("%d ", c[i]);

    return 0;
}
