int main() {
    int a[20], n, key, i, c=0, low=0, high, mid;
    printf("Enter n: ");
    scanf("%d",&n);
    printf("Enter sorted elements: ");

    scanf("%d",&a[i]);
    printf("Enter key: ");
    scanf("%d",&key);

    for(i=0;i<n;i++) {
        c++;
        if(a[i]==key) break;
    }
    printf("Linear comparisons=%d\n",c);
    if(i==n) printf("Not found\n");
    else printf("Found at %d\n",i);

    high=n-1; c=0;
    while(low<=high) {
        mid=(low+high)/2; c++;
        if(a[mid]==key) break;
        else if(key<a[mid]) high=mid-1;
        else low=mid+1;
    }
    printf("Binary comparisons=%d\n",c);
    if(low<=high) printf("Found at %d\n",mid);
    else printf("Not found\n");
    return 0;
}
