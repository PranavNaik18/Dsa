Q) Write a C program to implement Linear Search and Binary Search to search an element in an array and compare the number of comparisons.
#include <stdio.h>

int main()
{
    int a[50], n, key, i;
    int count1 = 0, count2 = 0;
    int low, high, mid, found;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements in sorted order:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter element to search: ");
    scanf("%d", &key);

    /* Linear Search */
    found = 0;

    for(i = 0; i < n; i++)
    {
        count1++;

        if(a[i] == key)
        {
            found = 1;
            break;
        }
    }

    if(found)
        printf("\nLinear Search: Found");
    else
        printf("\nLinear Search: Not Found");

    printf("\nComparisons = %d", count1);

    /* Binary Search */
    low = 0;
    high = n - 1;
    found = 0;

    while(low <= high)
    {
        mid = (low + high) / 2;
        count2++;

        if(a[mid] == key)
        {
            found = 1;
            break;
        }
        else if(key < a[mid])
            high = mid - 1;
        else
            low = mid + 1;
    }

    if(found)
        printf("\nBinary Search: Found");
    else
        printf("\nBinary Search: Not Found");

    printf("\nComparisons = %d", count2);

    return 0;
}






1) Design a student database to search records using USN.

#include <stdio.h>
#include <string.h>

struct Student
{
    char usn[20];
    char name[30];
};

int main()
{
    struct Student s[50];
    int n, i;
    char searchUSN[20];

    printf("Enter number of students: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("\nEnter USN: ");
        scanf("%s", s[i].usn);

        printf("Enter name: ");
        scanf("%s", s[i].name);
    }

    printf("\nEnter USN to search: ");
    scanf("%s", searchUSN);

    for(i = 0; i < n; i++)
    {
        if(strcmp(s[i].usn, searchUSN) == 0)
        {
            printf("Student Found\n");
            printf("Name = %s\n", s[i].name);
            return 0;
        }
    }

    printf("Student Not Found");

    return 0;
}

4)Implement Binary Search recursively and compare it with the iterative version. 

#include <stdio.h>

int binarySearch(int a[], int low, int high, int key)
{
    int mid;

    if(low > high)
        return -1;

    mid = (low + high) / 2;

    if(a[mid] == key)
        return mid;

    if(key < a[mid])
        return binarySearch(a, low, mid - 1, key);

    return binarySearch(a, mid + 1, high, key);
}

int main()
{
    int a[50], n, key, i, result;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter sorted elements:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter element: ");
    scanf("%d", &key);

    result = binarySearch(a, 0, n - 1, key);

    if(result == -1)
        printf("Not Found");
    else
        printf("Found at position %d", result + 1);

    return 0;
}
